cmake_minimum_required(VERSION 3.20)
project(ds4_store_test LANGUAGES CXX)

find_package(Threads REQUIRED)
add_executable(ds4_store_test ../ds4_store_test.cpp)
target_compile_features(ds4_store_test PRIVATE cxx_std_17)
target_include_directories(ds4_store_test PRIVATE ../../src/ds4 ../../include)
target_link_libraries(ds4_store_test PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(ds4_store_test PRIVATE /W4)
  target_compile_definitions(ds4_store_test PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(ds4_store_test PRIVATE -Wall -Wextra -Werror)
endif()
if(WIN32)
  target_compile_definitions(ds4_store_test PRIVATE _WIN32_WINNT=0x0A00)
endif()

enable_testing()
add_test(NAME ds4_store_test COMMAND ds4_store_test ${CMAKE_CURRENT_BINARY_DIR})
set_tests_properties(ds4_store_test PROPERTIES TIMEOUT 120)
